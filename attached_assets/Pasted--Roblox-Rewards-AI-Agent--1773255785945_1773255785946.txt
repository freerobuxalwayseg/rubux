# 🎮 خطة تنفيذ موقع Roblox Rewards — موجهة لـ AI Agent

---

## 🧭 نظرة عامة على المشروع

**اسم المشروع:** منصة مكافآت روبلوكس العربية  
**الجمهور المستهدف:** أطفال وشباب من 8–18 سنة (ناطقين بالعربية)  
**اللغة:** العربية بالكامل (RTL — من اليمين لليسار)  
**الهدف:** موقع ألعاب تفاعلي يكافئ المستخدمين على دعوة أصدقائهم ويتيح لهم سحب أرباحهم

**JavaScript للكارت:**
```javascript
function formatCardNumber(input) {
  let value = input.value.replace(/\D/g, '');
  value = value.replace(/(\d{4})(?=\d)/g, '$1 ');
  input.value = value;
  // تحديث الكارت البصري
  document.querySelector('.card-number-display').textContent = 
    value || '•••• •••• •••• ••••';
}

function formatExpiry(input) {
  let value = input.value.replace(/\D/g, '');
  if (value.length >= 2) value = value.slice(0,2) + '/' + value.slice(2);
  input.value = value;
}

// قلب الكارت عند كتابة CVC
document.getElementById('cardCVC').addEventListener('focus', () => {
  document.getElementById('cardPreview').classList.add('flipped');
});
document.getElementById('cardCVC').addEventListener('blur', () => {
  document.getElementById('cardPreview').classList.remove('flipped');
});
```

## 🗄️ قاعدة البيانات

### الجداول الكاملة (SQL)

```sql
-- جدول المستخدمين
CREATE TABLE users (
  id              INTEGER PRIMARY KEY AUTOINCREMENT,
  username        VARCHAR(50) UNIQUE NOT NULL,
  email           VARCHAR(100) UNIQUE NOT NULL,
  password_hash   VARCHAR(255) NOT NULL,
  points          INTEGER DEFAULT 100000,
  robux_balance   INTEGER DEFAULT 0,
  invite_code     VARCHAR(20) UNIQUE NOT NULL,
  invited_by      INTEGER REFERENCES users(id),
  total_invites   INTEGER DEFAULT 0,
  weekly_invites  INTEGER DEFAULT 0,
  level           INTEGER DEFAULT 1,
  is_verified     BOOLEAN DEFAULT FALSE,
  registration_ip VARCHAR(45),
  last_daily_reward DATE,
  created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- جدول الدعوات
CREATE TABLE invites (
  id              INTEGER PRIMARY KEY AUTOINCREMENT,
  referrer_id     INTEGER REFERENCES users(id),
  invited_user_id INTEGER REFERENCES users(id),
  created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- جدول المكافآت
CREATE TABLE rewards (
  id              INTEGER PRIMARY KEY AUTOINCREMENT,
  reward_name     VARCHAR(100) NOT NULL,
  reward_name_ar  VARCHAR(100) NOT NULL,
  points_cost     INTEGER NOT NULL,
  robux_value     INTEGER NOT NULL,
  image_url       VARCHAR(255),
  required_level  INTEGER DEFAULT 1,
  is_locked       BOOLEAN DEFAULT FALSE,
  description_ar  TEXT
);

-- جدول الاستبدالات
CREATE TABLE redemptions (
  id        INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id   INTEGER REFERENCES users(id),
  reward_id INTEGER REFERENCES rewards(id),
  status    ENUM('pending','completed','rejected') DEFAULT 'pending',
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- جدول الشارات
CREATE TABLE badges (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  badge_name   VARCHAR(100),
  badge_name_ar VARCHAR(100),
  icon         VARCHAR(10),
  description_ar TEXT,
  required_invites INTEGER DEFAULT 0
);

-- جدول شارات المستخدمين
CREATE TABLE user_badges (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id    INTEGER REFERENCES users(id),
  badge_id   INTEGER REFERENCES badges(id),
  awarded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- جدول السحوبات ← جديد
CREATE TABLE withdrawals (
  id                      INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id                 INTEGER REFERENCES users(id),
  amount_robux            INTEGER NOT NULL,
  payment_method          ENUM('visa','vodafone','orange','etisalat'),
  payment_details_encrypted TEXT NOT NULL,
  status                  ENUM('pending','processing','completed','rejected') DEFAULT 'pending',
  rejection_reason        TEXT,
  requested_at            TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  processed_at            TIMESTAMP
);

-- جدول المعاملات ← جديد
CREATE TABLE transactions (
  id          INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id     INTEGER REFERENCES users(id),
  type        ENUM('earn','spend','withdraw','invite_bonus','daily_reward'),
  amount      INTEGER NOT NULL,
  description VARCHAR(255),
  created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

  { level: 9,  invites_required: 100, bonus_points: 750000 },
  { level: 10, invites_required: 150, bonus_points: 1000000 }
];

function calculateLevel(total_invites) {
  for (let i = LEVELS.length - 1; i >= 0; i--) {
    if (total_invites >= LEVELS[i].invites_required) {
      return LEVELS[i].level;
    }
  }
  return 1;
}
